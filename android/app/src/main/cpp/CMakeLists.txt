cmake_minimum_required(VERSION 3.22)
project(rp)
set(ROOT ${CMAKE_CURRENT_SOURCE_DIR}/../../../../..)
set(FF ${ROOT}/build/ffmpeg-android/${ANDROID_ABI})
add_library(rp SHARED rp_jni.c ${ROOT}/Core/premo.c ${ROOT}/Core/rp_player.c)
target_compile_definitions(rp PRIVATE RP_PIXFMT=AV_PIX_FMT_RGBA)
target_include_directories(rp PRIVATE ${ROOT}/Core ${FF}/include)
target_link_libraries(rp ${FF}/lib/libavcodec.a ${FF}/lib/libswscale.a ${FF}/lib/libswresample.a ${FF}/lib/libavutil.a log m z)
target_link_options(rp PRIVATE "-Wl,-Bsymbolic")
